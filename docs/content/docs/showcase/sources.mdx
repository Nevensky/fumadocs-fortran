!> @file modern_doxygen.f90
!> @brief Doxygen-style documentation, as used by LAPACK-derived and many HPC codes.
!>
!> @details The module wraps a few BLAS/LAPACK calls.

!> @brief Eigen-solvers.
!> @author A. Person
module eigen_wrappers
  use iso_fortran_env, only: wp => real64
  implicit none
  private
  public :: eigh, gemm_wrap

  integer, parameter :: max_iter = 100 !< iteration cap (Doxygen post-comment)

contains

  !> @brief Eigenvalues of a Hermitian matrix with LAPACK zheevd: \f$ A v = \lambda v \f$.
  !>
  !> @details Diagonalises the matrix in place by divide and conquer,
  !> \f[ A = V \Lambda V^{H}, \qquad \Lambda = \mathrm{diag}(w_1, \dots, w_n), \f]
  !> where the columns of \f$V\f$ are the orthonormal eigenvectors.
  !>
  !> @param[in,out] a    on entry the Hermitian matrix; on exit the eigenvectors
  !> @param[out]    w    the eigenvalues in ascending order, $w_1 \le w_2 \le \dots \le w_n$
  !> @param[out]    info LAPACK info code: 0 on success
  !> @returns nothing; it is a subroutine
  !> @see zheevd
  subroutine eigh(a, w, info)
    complex(wp), intent(inout) :: a(:,:)
    real(wp),    intent(out)   :: w(:)
    integer,     intent(out)   :: info
    complex(wp), allocatable :: work(:)
    real(wp),    allocatable :: rwork(:)
    integer,     allocatable :: iwork(:)
    integer :: n, lwork, lrwork, liwork
    n = size(a, 1)
    lwork = 2*n + n*n
    lrwork = 1 + 5*n + 2*n*n
    liwork = 3 + 5*n
    allocate(work(lwork), rwork(lrwork), iwork(liwork))
    call zheevd('V', 'U', n, a, n, w, work, lwork, rwork, lrwork, iwork, liwork, info)
  end subroutine eigh

  !> \brief \f$ C \leftarrow \alpha A B + \beta C \f$ with BLAS dgemm.
  !> \param[in] alpha scale of the product
  !> \param[in] a left factor
  !> \param[in] b right factor
  !> \param[in] beta scale of C
  !> \param[in,out] c accumulator
  subroutine gemm_wrap(alpha, a, b, beta, c)
    real(wp), intent(in)    :: alpha   !< scale of A*B
    real(wp), intent(in)    :: a(:,:)  !< left factor
    real(wp), intent(in)    :: b(:,:)  !< right factor
    real(wp), intent(in)    :: beta    !< scale of C
    real(wp), intent(inout) :: c(:,:)  !< accumulator
    call dgemm('N', 'N', size(a,1), size(b,2), size(a,2), alpha, a, size(a,1), &
               b, size(b,1), beta, c, size(c,1))
  end subroutine gemm_wrap

end module eigen_wrappers

!> Dense and banded linear-algebra kernels for a small solver: matrix types, products threaded with OpenMP and
!> offloaded with OpenACC, norms reduced over MPI ranks, and LAPACK-backed solves.
module solver_kernels
  !! The kernels solve \( A x = b \) for a matrix \(A \in \mathbb{R}^{n \times n}\), by LU factorisation with
  !! partial pivoting:
  !!
  !! $$ P A = L U, \qquad x = U^{-1} L^{-1} P b. $$
  !!
  !! The residual they report is $r = \|b - A x\|_2 / \|b\|_2$, and the condition number estimate
  !! \begin{equation}
  !!   \kappa_1(A) = \|A\|_1 \, \|A^{-1}\|_1
  !! \end{equation}
  !! comes from LAPACK's `dgecon`.
  !!
  !! @note
  !! Residuals are reported in the 2-norm, which the maximum norm bounds:
  !! \[ \begin{aligned} \|x\|_2^2 &= \textstyle\sum_i x_i^2 \\ &\le n \max_i x_i^2 \end{aligned} \]
  !! @endnote
  !!
  !! Start with [[dense_matrix]] and the generic [[solve_dense]].
  use, intrinsic :: iso_fortran_env, only: dp => real64, int64
  use, intrinsic :: iso_c_binding, only: c_double, c_int
#ifdef _OPENMP
  use omp_lib, only: omp_get_max_threads
#endif
#ifdef USE_MPI
  use mpi_f08, only: MPI_Comm, MPI_Allreduce, MPI_IN_PLACE, MPI_SUM, MPI_DOUBLE_PRECISION
#endif
  implicit none
  private

  public :: dense_matrix, banded_matrix, solve_dense, norm2_parallel
  public :: operator(+), assignment(=), matrix_free
  public :: color, RED, GREEN

  integer, parameter, public :: max_order = 64 !! largest supported order

  enum, bind(c)
    enumerator :: RED = 1, GREEN
  end enum

  type :: color
    integer :: rgb(3) = 0 !! red, green and blue channels
  end type color

  !> An abstract matrix type.
  type, abstract, public :: abstract_matrix
    integer :: n = 0 !! order of the matrix
  contains
    procedure(matvec_iface), deferred, pass(self) :: matvec !! y = A x
    procedure :: order => matrix_order
    generic :: write(formatted) => write_matrix
    procedure, private :: write_matrix
  end type abstract_matrix

  abstract interface
    !> Matrix-vector product interface: $y \leftarrow A x$.
    subroutine matvec_iface(self, x, y)
      import :: abstract_matrix, dp
      class(abstract_matrix), intent(in) :: self
      real(dp), intent(in)  :: x(:)  !! input vector
      real(dp), intent(out) :: y(:)  !! output vector
    end subroutine matvec_iface
  end interface

  !> Dense, column-major matrix.
  type, extends(abstract_matrix) :: dense_matrix
    real(dp), allocatable :: a(:,:) !! the entries
#ifdef USE_MPI
    type(MPI_Comm) :: comm !! communicator the matrix is distributed over
#endif
  contains
    procedure, pass(self) :: matvec => dense_matvec
    final :: dense_finalize
  end type dense_matrix

  !> A parameterised derived type: banded storage, where $a_{ij} = 0$ for $j < i - k_l$ or $j > i + k_u$, and
  !> $a_{ij}$ is kept in `ab(ku+1+i-j, j)`.
  type, extends(abstract_matrix) :: banded_matrix(k, kl, ku)
    integer, kind :: k = dp      !! real kind
    integer, len  :: kl = 1      !! sub-diagonals
    integer, len  :: ku = 1      !! super-diagonals
    real(k) :: ab(kl+ku+1, 100)  !! band storage, LAPACK layout
  contains
    procedure, pass(self) :: matvec => banded_matvec
  end type banded_matrix

  interface operator(+)
    !! Add two dense matrices.
    module procedure add_dense
  end interface

  interface assignment(=)
    module procedure assign_from_array
  end interface

  !> Generic interface with module procedures implemented in a submodule.
  interface solve_dense
    module subroutine solve_dense_real(a, b, info)
      !! Solve \( A X = B \) with LAPACK `dgesv`.
      real(dp), intent(inout) :: a(:,:)        !! on exit, the LU factors
      real(dp), intent(inout) :: b(:,:)        !! right-hand sides; on exit, the solution
      integer,  intent(out), optional :: info  !! LAPACK info code
    end subroutine solve_dense_real
    module subroutine solve_dense_complex(a, b, info)
      !! Complex version, calls `zgesv`.
      complex(dp), intent(inout) :: a(:,:)
      complex(dp), intent(inout) :: b(:,:)
      integer,     intent(out), optional :: info
    end subroutine solve_dense_complex
  end interface solve_dense

  interface
    !> An external C routine.
    function matrix_free(n, x) result(r) bind(c, name="matrix_free_c")
      import :: c_int, c_double
      integer(c_int), value, intent(in) :: n
      real(c_double), intent(in) :: x(n)
      real(c_double) :: r
    end function matrix_free
  end interface

contains

  !> Order of a matrix.
  pure integer function matrix_order(self) result(n)
    class(abstract_matrix), intent(in) :: self
    n = self%n
  end function matrix_order

  subroutine write_matrix(self, unit, iotype, v_list, iostat, iomsg)
    class(abstract_matrix), intent(in) :: self
    integer, intent(in) :: unit
    character(*), intent(in) :: iotype
    integer, intent(in) :: v_list(:)
    integer, intent(out) :: iostat
    character(*), intent(inout) :: iomsg
    write(unit, '(a,i0)', iostat=iostat, iomsg=iomsg) 'matrix of order ', self%n
  end subroutine write_matrix

  subroutine dense_matvec(self, x, y)
    !! Dense product $y = A x$, threaded with OpenMP when available, then again with BLAS `dgemv`,
    !! which computes $y \leftarrow \alpha A x + \beta y$ (here $\alpha = 1$, $\beta = 0$).
    class(dense_matrix), intent(in) :: self
    real(dp), intent(in)  :: x(:)
    real(dp), intent(out) :: y(:)
    integer :: i
    !$omp parallel do default(shared) private(i) schedule(static)
    do i = 1, self%n
      y(i) = dot_product(self%a(i,:), x)
    end do
    !$omp end parallel do
    call dgemv('N', self%n, self%n, 1.0_dp, self%a, self%n, x, 1, 0.0_dp, y, 1)
  end subroutine dense_matvec

  subroutine banded_matvec(self, x, y)
    class(banded_matrix(k=dp, kl=*, ku=*)), intent(in) :: self
    real(dp), intent(in)  :: x(:)
    real(dp), intent(out) :: y(:)
    call dgbmv('N', self%n, self%n, self%kl, self%ku, 1.0_dp, self%ab, self%kl+self%ku+1, x, 1, 0.0_dp, y, 1)
  end subroutine banded_matvec

  impure elemental subroutine dense_finalize(self)
    type(dense_matrix), intent(inout) :: self
    if (allocated(self%a)) deallocate(self%a)
  end subroutine dense_finalize

  function add_dense(a, b) result(c)
    type(dense_matrix), intent(in) :: a, b
    type(dense_matrix) :: c
    c%n = a%n
    c%a = a%a + b%a
  end function add_dense

  subroutine assign_from_array(lhs, rhs)
    type(dense_matrix), intent(out) :: lhs
    real(dp), intent(in) :: rhs(:,:)
    lhs%n = size(rhs, 1)
    lhs%a = rhs
  end subroutine assign_from_array

  !> 2-norm reduced across OpenMP threads and MPI ranks: each rank $r$ sums the squares of its slice
  !> $I_r$, and an `MPI_Allreduce` adds the partial sums,
  !> \[ \|x\|_2 = \Big( \sum_{r=0}^{p-1} \sum_{i \in I_r} x_i^2 \Big)^{1/2}. \]
  function norm2_parallel(x &
#ifdef USE_MPI
      , comm &
#endif
      ) result(nrm)
    real(dp), intent(in) :: x(:) !! local slice of the vector
#ifdef USE_MPI
    type(MPI_Comm), intent(in) :: comm !! communicator
#endif
    real(dp) :: nrm !! the global 2-norm
    real(dp) :: s
    integer :: i
    s = 0.0_dp
    !$omp parallel do reduction(+:s)
    do i = 1, size(x)
      s = s + x(i)**2
    end do
    !$omp end parallel do
#ifdef USE_MPI
    call MPI_Allreduce(MPI_IN_PLACE, s, 1, MPI_DOUBLE_PRECISION, MPI_SUM, comm)
#endif
    nrm = sqrt(s)
  end function norm2_parallel

  !> A device routine for OpenACC and OpenMP offload: one element of $a x + y$.
  pure real(dp) function axpy_elem(a, x, y)
    !$acc routine seq
    !$omp declare target
    real(dp), value :: a, x, y
    axpy_elem = a*x + y
  end function axpy_elem

  !> Rank-generic sum $s = \sum_{i} x_i$ with `select rank`, reduced over all images with `co_sum`.
  subroutine rank_sum(x, total)
    real(dp), intent(in) :: x(..)
    real(dp), intent(out) :: total[*]
    select rank (x)
    rank (1)
      total = sum(x)
    rank (2)
      total = sum(x)
    rank default
      total = 0.0_dp
    end select
    call co_sum(total)
  end subroutine rank_sum

  !> Scales a vector in place,
  !> $$ x_i \leftarrow \alpha \, x_i, \qquad i = 1, \dots, n, $$
  !> in a `do concurrent` loop, where `local_init` gives each iteration its own copy of the factor.
  subroutine concurrent_scale(x, alpha)
    real(dp), intent(inout) :: x(:)
    real(dp), intent(in) :: alpha
    integer :: i
    real(dp) :: s
    s = alpha
    do concurrent (i = 1:size(x)) local_init(s)
      x(i) = s*x(i)
    end do
  end subroutine concurrent_scale

end module solver_kernels

!> Submodule with the LAPACK-backed implementations.
submodule (solver_kernels) solver_kernels_lapack
  implicit none
contains
  module subroutine solve_dense_real(a, b, info)
    real(dp), intent(inout) :: a(:,:)
    real(dp), intent(inout) :: b(:,:)
    integer,  intent(out), optional :: info
    integer :: ipiv(size(a,1)), stat
    call dgesv(size(a,1), size(b,2), a, size(a,1), ipiv, b, size(b,1), stat)
    if (present(info)) info = stat
  end subroutine solve_dense_real

  module procedure solve_dense_complex
    integer :: ipiv(size(a,1)), stat
    call zgesv(size(a,1), size(b,2), a, size(a,1), ipiv, b, size(b,1), stat)
    if (present(info)) info = stat
  end procedure solve_dense_complex
end submodule solver_kernels_lapack

!> A driver program.

*> \brief \b DTRIDS solves a tridiagonal system by the Thomas algorithm
*
*  Definition:
*  ===========
*
*       SUBROUTINE DTRIDS( N, NRHS, DL, D, DU, B, LDB, INFO )
*
*       .. Scalar Arguments ..
*       INTEGER            INFO, LDB, N, NRHS
*       ..
*       .. Array Arguments ..
*       DOUBLE PRECISION   B( LDB, * ), D( * ), DL( * ), DU( * )
*       ..
*
*
*> \par Purpose:
*  =============
*>
*> \verbatim
*>
*> DTRIDS solves A * X = B, where A is an N-by-N tridiagonal matrix,
*> by Gaussian elimination without pivoting (the Thomas algorithm).
*> It is written for the fumadocs-fortran example in the style of
*> Reference-LAPACK; use DGTSV from LAPACK for real work.
*> \endverbatim
*
*  Arguments:
*  ==========
*
*> \param[in] N
*> \verbatim
*>          N is INTEGER
*>          The order of the matrix A.  N >= 0.
*> \endverbatim
*>
*> \param[in] NRHS
*> \verbatim
*>          NRHS is INTEGER
*>          The number of right hand sides, i.e., the number of columns
*>          of the matrix B.  NRHS >= 0.
*> \endverbatim
*>
*> \param[in] DL
*> \verbatim
*>          DL is DOUBLE PRECISION array, dimension (N-1)
*>          The (n-1) sub-diagonal elements of A.
*> \endverbatim
*>
*> \param[in,out] D
*> \verbatim
*>          D is DOUBLE PRECISION array, dimension (N)
*>          On entry, the diagonal elements of A.
*>          On exit, the diagonal elements of the upper triangular
*>          factor U.
*> \endverbatim
*>
*> \param[in] DU
*> \verbatim
*>          DU is DOUBLE PRECISION array, dimension (N-1)
*>          The (n-1) super-diagonal elements of A.
*> \endverbatim
*>
*> \param[in,out] B
*> \verbatim
*>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
*>          On entry, the N-by-NRHS right hand side matrix B.
*>          On exit, if INFO = 0, the N-by-NRHS solution matrix X.
*> \endverbatim
*>
*> \param[in] LDB
*> \verbatim
*>          LDB is INTEGER
*>          The leading dimension of the array B.  LDB >= max(1,N).
*> \endverbatim
*>
*> \param[out] INFO
*> \verbatim
*>          INFO is INTEGER
*>          = 0:  successful exit
*>          < 0:  if INFO = -i, the i-th argument had an illegal value
*>          > 0:  if INFO = i, U(i,i) is exactly zero, and the solution
*>                has not been computed.
*> \endverbatim
*
*  Authors:
*  ========
*
*> \author The fumadocs-fortran example
*
*> \ingroup gtsv
*
*  =====================================================================
      SUBROUTINE DTRIDS( N, NRHS, DL, D, DU, B, LDB, INFO )
*
*     .. Scalar Arguments ..
      INTEGER            INFO, LDB, N, NRHS
*     ..
*     .. Array Arguments ..
      DOUBLE PRECISION   B( LDB, * ), D( * ), DL( * ), DU( * )
*     ..
*
*  =====================================================================
*
*     .. Parameters ..
      DOUBLE PRECISION   ZERO
      PARAMETER          ( ZERO = 0.0D+0 )
*     ..
*     .. Local Scalars ..
      INTEGER            I, J
      DOUBLE PRECISION   FACT
*     ..
*     .. External Subroutines ..
      EXTERNAL           XERBLA
*     ..
*     .. Intrinsic Functions ..
      INTRINSIC          MAX
*     ..
*     .. Executable Statements ..
*
      INFO = 0
      IF( N.LT.0 ) THEN
         INFO = -1
      ELSE IF( NRHS.LT.0 ) THEN
         INFO = -2
      ELSE IF( LDB.LT.MAX( 1, N ) ) THEN
         INFO = -7
      END IF
      IF( INFO.NE.0 ) THEN
         CALL XERBLA( 'DTRIDS', -INFO )
         RETURN
      END IF
      IF( N.EQ.0 ) RETURN
*
*     Forward elimination
*
      DO 20 I = 1, N - 1
         IF( D( I ).EQ.ZERO ) THEN
            INFO = I
            RETURN
         END IF
         FACT = DL( I ) / D( I )
         D( I+1 ) = D( I+1 ) - FACT*DU( I )
         DO 10 J = 1, NRHS
            B( I+1, J ) = B( I+1, J ) - FACT*B( I, J )
   10    CONTINUE
   20 CONTINUE
      IF( D( N ).EQ.ZERO ) THEN
         INFO = N
         RETURN
      END IF
*
*     Back substitution
*
      DO 40 J = 1, NRHS
         B( N, J ) = B( N, J ) / D( N )
         DO 30 I = N - 1, 1, -1
            B( I, J ) = ( B( I, J )-DU( I )*B( I+1, J ) ) / D( I )
   30    CONTINUE
   40 CONTINUE
      RETURN
*
*     End of DTRIDS
*
      END
